% Les ptites dependences
\usepackage{graphicx}

\usepackage{float}
\usepackage{caption}
\usepackage{calc}


% ==============================================================
% \image<legende>[taille]{chemin}
%
%   <legende> (optionnel, delimite par < >)
%             texte affiche sous l'image via \caption
%   [taille]  (optionnel, delimite par [ ]), defaut : 0.8
%             fraction de \linewidth occupee par l'image
%   {chemin}  (obligatoire) chemin vers le fichier image
%
%   Exemple :
%       \image<Photo d'un pingouin>[0.6]{pingouin.png}
%       \image{pingouin.png}                (legende et taille par defaut)
% ==============================================================
% Tu peux redefinir la taille par defaut ici
\NewDocumentCommand{\image}{d<>O{0.5}m}{%
    \begin{figure}[H]
        \centering
        \includegraphics[width=#2\linewidth]{#3}%
        \IfNoValueF{#1}{\caption{#1}}%
    \end{figure}%
}


% ==============================================================
% \tableau<legende>[nb colonnes]{contenu}
%
%   <legende>      (optionnel, delimite par < >)
%                  texte affiche via \caption
%   [nb colonnes]  (optionnel, delimite par [ ]), defaut : 2
%                  nombre de colonnes du tableau
%   {contenu}      (obligatoire) lignes du tableau, colonnes
%                  separees par & et lignes terminees par \\
%
%   Exemple :
%       \tableau<Donnees de l'experience abc>[2]{
%           echantillon & force \\
%           1 & 2 \\
%           3 & 4 \\
%       }
% ==============================================================
\NewDocumentCommand{\tableau}{d<>O{2}m}{%
    \begin{table}[H]
        \centering
        \begin{tabular}{|*{#2}{c|}}
            \hline
            #3
            \hline
        \end{tabular}%
        \IfNoValueF{#1}{\caption{#1}}%
    \end{table}%
}


% ==============================================================
% \split[ratio]{gauche}{droite}
%
%   [ratio]   (optionnel, delimite par [ ]), defaut : 0.5
%             largeur de la minipage de gauche, en fraction de
%             \linewidth (la minipage de droite prend le reste,
%             moins un petit espace de separation)
%   {gauche}  (obligatoire) contenu de la minipage de gauche
%   {droite}  (obligatoire) contenu de la minipage de droite
%
%   Exemple :
%       \split{Description d'un lion}{\image[0.8]{lion.png}}
%       \split[0.3]{texte court a gauche}{texte a droite}
% ==============================================================
\NewDocumentCommand{\split}{O{0.5}mm}{%
    \noindent
    \begin{minipage}[t]{#1\linewidth}
        #2
    \end{minipage}%
    \hfill
    \begin{minipage}[t]{\linewidth-#1\linewidth-1em}
        #3
    \end{minipage}%
}
