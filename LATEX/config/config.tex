% ==============================================================
%  Parametres ajustables
%
%  \big et \little fixent le rapport entre les deux moities dans
%  les commandes \showB.. (1ere moitie = \big, 2e = \little) et
%  \showL.. (1ere moitie = \little, 2e = \big). Ce sont des
%  fractions de \textwidth ; changez-les ici, ou ailleurs dans
%  le document avec \redefinir (voir plus bas), pour affecter
%  tous les \showB../\showL.. qui suivent.
%
%  ATTENTION : \big est deja une commande LaTeX standard (taille
%  de delimiteurs en mode mathematique, ex. \big( \big[ ). La
%  redefinir ici l'ecrase pour tout le document : si vous utilisez
%  \big( / \big[ / \big\{ ailleurs, renommez cette variable (par
%  exemple \bigratio / \littleratio) plutot que de garder \big.
%
%  Exemple :
%      \redefinir{\big}{0.65}
%      \redefinir{\little}{0.25}
%      \showBXI{Texte large.}{image.png}   % utilise le nouveau rapport
% ==============================================================
\renewcommand{\big}{0.55}
\newcommand{\little}{0.35}


% ==============================================================
% \image<legende>[taille]{chemin}<label>
%
%   <legende> (optionnel, delimite par < >)
%             texte affiche sous l'image via \caption
%   [taille]  (optionnel, delimite par [ ]), defaut : 0.8
%             fraction de \linewidth occupee par l'image
%   {chemin}  (obligatoire) chemin vers le fichier image
%
%   Exemple :
%       \image<Photo d'un pingouin>[0.6]{pingouin.png}
%       \image{pingouin.png}                (legende et taille par defaut)
% ==============================================================
\NewDocumentCommand{\image}{d<>O{0.4}md<>}{%
    \begin{figure}[H]
        \centering
        \includegraphics[width=#2\linewidth]{#3}%
        \IfNoValueF{#1}{\caption{#1}}%
        \IfNoValueF{#4}{\label{#4}}%
    \end{figure}%
}


% ==============================================================
% \tableau<legende>[nb colonnes]{contenu}<label>
%
%   <legende>      (optionnel, delimite par < >)
%                  texte affiche via \caption
%   [nb colonnes]  (optionnel, delimite par [ ]), defaut : 2
%                  nombre de colonnes du tableau
%   {contenu}      (obligatoire) lignes du tableau, colonnes
%                  separees par & et lignes terminees par \\
%
%   Exemple :
%       \tableau<Donnees de l'experience abc>[2]{
%           echantillon & force \\
%           1 & 2 \\
%           3 & 4 \\
%       }
% ==============================================================
\NewDocumentCommand{\tableau}{d<>O{2}md<>}{%
    \begin{table}[H]
        \centering
        \begin{tabular}{|*{#2}{c|}}
            \hline
            #3
            \hline
        \end{tabular}%
        \IfNoValueF{#1}{\caption{#1}}%
        \IfNoValueF{#4}{\label{#4}}%
    \end{table}%
}


% ==============================================================
% \split[ratio]{gauche}{droite}
%
%   [ratio]   (optionnel, delimite par [ ]), defaut : 0.5
%             largeur de la minipage de gauche, en fraction de
%             \linewidth (la minipage de droite prend le reste,
%             moins un petit espace de separation)
%   {gauche}  (obligatoire) contenu de la minipage de gauche
%   {droite}  (obligatoire) contenu de la minipage de droite
%
%   Exemple :
%       \split{Description d'un lion}{\image[0.8]{lion.png}}
%       \split[0.3]{texte court a gauche}{texte a droite}
% ==============================================================
\NewDocumentCommand{\split}{O{0.5}mm}{%
    \noindent
    \begin{minipage}[t]{#1\linewidth}
        #2
    \end{minipage}%
    \hfill
    \begin{minipage}[t]{\linewidth-#1\linewidth-1em}
        #3
    \end{minipage}%
}


% ==============================================================
% \note{texte}
%
%   {texte}  (obligatoire) texte a mettre en avant, centre et en gras
%
%   Exemple :
%       \note{Attention : ne pas depasser 50 degres.}
% ==============================================================
\NewDocumentCommand{\note}{m}{%
    \begin{center}
        \textbf{#1}
    \end{center}%
}


% ==============================================================
% \sidenote[r|l]{contenu}
%
%   [r|l]      (optionnel, delimite par [ ]), defaut : r
%              cote de la marge ou placer la note : r = droite,
%              n'importe quelle autre valeur (ex. l) = gauche
%   {contenu}  (obligatoire) texte a placer en marge
%
%   Repose sur \marginnote (package marginnote) plutot que sur
%   \marginpar : le cote est decide immediatement a l'appel (via
%   \reversemarginpar / \normalmarginpar dans un groupe local),
%   ce qui permet d'alterner gauche/droite librement, y compris
%   plusieurs fois sur la meme page (\marginpar seul ne le permet
%   pas correctement, la decision du cote etant retardee par
%   LaTeX jusqu'a la construction de la page).
%
%   Exemple :
%       \sidenote[r]{Remarque a droite.}
%       \sidenote[l]{Remarque a gauche.}
%       \sidenote{Remarque a droite (par defaut).}
% ==============================================================
\NewDocumentCommand{\sidenote}{O{r}m}{%
    \begingroup
    \if r#1%
        \normalmarginpar
    \else
        \reversemarginpar
    \fi
    \marginnote{\footnotesize #2}%
    \endgroup
}


% ==============================================================
% \definesivide{\macro}{valeur par defaut}
%
%   {\macro}          (obligatoire) la macro/variable a verifier
%   {valeur defaut}   (obligatoire) valeur utilisee si \macro est
%                      vide ou non definie
%
%   Si \macro est vide (ou n'existe pas encore), elle est
%   (re)definie avec la valeur par defaut. Sinon, elle est
%   laissee inchangee.
%
%   NB : utilise des commandes internes LaTeX (@ dans le nom),
%   d'ou le \makeatletter / \makeatother autour de la definition.
%
%   Exemple :
%       \def\titre{}
%       \definesivide{\titre}{Sans titre}
%       \titre   % affiche "Sans titre"
%
%       \def\auteur{Jean Dupont}
%       \definesivide{\auteur}{Anonyme}
%       \auteur  % affiche "Jean Dupont" (inchange)
% ==============================================================
\makeatletter
\NewDocumentCommand{\definesivide}{mm}{%
    \@ifundefined{\expandafter\@gobble\string#1}{%
        \def#1{#2}%
    }{%
        \ifx#1\@empty
            \def#1{#2}%
        \fi
    }%
}
\makeatother


% ==============================================================
% \redefinir{\macro}{nouvelle valeur}
%
%   {\macro}          (obligatoire) la macro/variable a definir
%   {nouvelle valeur} (obligatoire) nouvelle valeur de la macro
%
%   (Re)definit \macro avec la nouvelle valeur, que \macro ait
%   deja ete definie ou non. Contrairement a \newcommand (echoue
%   si \macro existe deja) ou \renewcommand (echoue si \macro
%   n'existe pas encore), \redefinir fonctionne dans tous les cas.
%
%   Exemple :
%       \def\titre{Ancien titre}
%       \redefinir{\titre}{Nouveau titre}
%       \titre   % affiche "Nouveau titre"
% ==============================================================
\NewDocumentCommand{\redefinir}{mm}{%
    \def#1{#2}%
}


% ==============================================================
%  Commandes \show.. : image(s) et/ou texte cote a cote
%
%  Chaque image peut recevoir une legende optionnelle, ajoutee
%  juste AVANT son argument obligatoire, avec la syntaxe < >
%  (comme pour \image). La legende est affichee sous l'image via
%  \captionof{figure}{...} (fourni par le package caption, deja
%  charge plus haut), ce qui fonctionne meme hors d'un flottant.
%  Un argument texte (X) n'a pas de legende, seules les images (I)
%  en ont une.
%
%  Convention de nommage (reprise telle quelle du code d'origine) :
%    I = image, X = texte
%    (sans prefixe) = les deux moities font 0.45\textwidth
%    B = 1ere moitie grande (\big), 2e petite (\little)
%    L = 1ere moitie petite (\little), 2e grande (\big)
%    (\big et \little sont definis plus haut, et redefinissables)
%
%  Exemples :
%      \showII{a.png}{b.png}
%      \showII<Legende a>{a.png}{b.png}
%      \showII{a.png}<Legende b>{b.png}
%      \showII<Legende a>{a.png}<Legende b>{b.png}
%      \showIX<Legende>{photo.png}{Un texte a droite.}
%      \showXI{Un texte a gauche.}<Legende>{photo.png}
% ==============================================================

% ---- image + texte (0.45 / 0.45) ----
\NewDocumentCommand{\showIX}{d<>md<>m}{%
    \begin{center}
        \begin{minipage}{0.45\textwidth}
            \centering
            \includegraphics[width=\textwidth]{#2}%
            \IfNoValueF{#1}{\captionof{figure}{#1}}%
            \IfNoValueF{#3}{\label{#3}}%
        \end{minipage}%
        \hfill
        \begin{minipage}{0.45\textwidth}
            #4
        \end{minipage}%
    \end{center}
}

% ---- texte + image (0.45 / 0.45) ----

            \includegraphics[width=\textwidth]{#3}%
            \IfNoValueF{#2}{\captionof{figure}{#2}}%
            \IfNoValueF{#4}{\label{#4}}%
        \end{minipage}%
    \end{center}
}
