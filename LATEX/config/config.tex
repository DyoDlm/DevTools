% ---------- Tableau seul, centré, avec légende ----------
\NewDocumentCommand{\showtab}{O{4} m m}{%
  \begin{table}[h]
    \centering
    \footnotesize
    \caption{#2}
    \begin{tblr}{
      colspec = {*{#1}{Q[c,m]}},
      hlines, vlines,
      rowhead = 1,
    }
      #3
    \end{tblr}
  \end{table}%
}

% ---------- Deux tableaux côte à côte, centrés, une légende commune ----------
\NewDocumentCommand{\showTABTAB}{O{4} m m m}{%
  \begin{table}[h]
    \centering
    \footnotesize
    \caption{#2}
    \hfill
    \begin{minipage}[t]{0.45\textwidth}
      \centering
      \vspace{0pt}
      \begin{tblr}{
        colspec = {*{#1}{Q[c,m]}},
        hlines, vlines,
        rowhead = 1,
      }
        #3
      \end{tblr}
    \end{minipage}%
    \hfill
    \begin{minipage}[t]{0.45\textwidth}
      \centering
      \vspace{0pt}
      \begin{tblr}{
        colspec = {*{#1}{Q[c,m]}},
        hlines, vlines,
        rowhead = 1,
      }
        #4
      \end{tblr}
    \end{minipage}%
    \hfill
  \end{table}%
}
