    %   #2 {ITEMS}

% ---------- Tableau seul, centré, avec légende ----------
\NewDocumentCommand{\showtab}{O{4} m m}{%
  \begin{table}[h]
    \centering
    \footnotesize
    \caption{#2}
    \begin{tblr}{
      colspec = {*{#1}{Q[c,m]}},
      hlines, vlines,
      rowhead = 1,
    }
    \\
      #3
    \\
    \end{tblr}
  \end{table}%
}

% ---------- Deux tableaux côte à côte, centrés, une légende commune (optionnelle) ----------
\NewDocumentCommand{\showTABTAB}{O{4} m m m}{%
  \begin{table}[h]
    \centering
    \footnotesize
    \caption{#2}
    \hfill
    \begin{minipage}[t]{0.45\textwidth}
      \centering
      \vspace{0pt}
      \begin{tblr}{
        colspec = {*{#1}{Q[c,m]}},
        hlines, vlines,
        rowhead = 1,
      }
        #3
      \end{tblr}
    \end{minipage}%
    \hfill
    \begin{minipage}[t]{0.45\textwidth}
      \centering
      \vspace{0pt}
      \begin{tblr}{
        colspec = {*{#1}{Q[c,m]}},
        hlines, vlines,
        rowhead = 1,
      }
        #4
      \end{tblr}
    \end{minipage}%
    \hfill
  \end{table}%
}

%=============================================================%
%   CMD :
    %   showI

%   DESCRIPTION :
    %   Format one image

%   ARGS :
    %   <CAPTION> (optional)
    %   #1 {SIZE}
    %   #2 {path_to_image}

%   Usage :
    %   \showI{size}{path_to_the_img}
    %   \showI<caption>{size}{path_to_the_img}

\NewDocumentCommand{\showI}{d<> m m}{%
  \begin{figure}[H]
    \centering
    \includegraphics[width=#2\textwidth]{#3}%
    \IfNoValueF{#1}{\caption{#1}}%
  \end{figure}%
}

%=============================================================%
%   CMD :
    %   showII

%   DESCRIPTION :
    %   IMG/IMG : SAME SIZE

%   ARGS
    %   #1 {IMG1} (size: 0.45)
    %   #2 {IMG2} (size: 0.45)

%   Usage : \showII{path_to_1}{path_to_2}

\NewDocumentCommand{\showII}{d<> m d<> m}{%
  \begin{center}
    \begin{minipage}{0.45\textwidth}
      \centering
      \includegraphics[width=\textwidth]{#2}\\[0.3em]
      \IfNoValueF{#1}{\small #1}
    \end{minipage}%
    \hfill
    \begin{minipage}{0.45\textwidth}
      \centering
      \includegraphics[width=\textwidth]{#4}\\[0.3em]
      \IfNoValueF{#3}{\small #3}
    \end{minipage}
  \end{center}%
}
