
\usepackage{xparse}

% ==============================================================
% \definesivide{\macro}{valeur par defaut}
%
%   {\macro}          (obligatoire) la macro/variable a verifier
%   {valeur defaut}   (obligatoire) valeur utilisee si \macro est vide
%                      ou non definie
%
%   Si \macro est vide (ou n'existe pas encore), elle est
%   (re)definie avec la valeur par defaut. Sinon, elle est
%   laissee inchangee.
%
%   Exemple :
%       \def\titre{}
%       \definesivide{\titre}{Sans titre}
%       \titre   % affiche "Sans titre"
%
%       \def\auteur{Jean Dupont}
%       \definesivide{\auteur}{Anonyme}
%       \auteur  % affiche "Jean Dupont" (inchange)
% ==============================================================
\NewDocumentCommand{\define}{mm}{%
    \@ifundefined{\expandafter\@gobble\string#1}{%
        \def#1{#2}%
    }{%
        \ifx#1\@empty
            \def#1{#2}%
        \fi
    }%
}

\NewDocumentCommand{\redefine}{mm}{%
    \def#1{#2}%
}
